const express = require("express"); const http = require("http"); const { Server } = require("socket.io"); const sqlite3 = require("sqlite3").verbose(); const path = require("path"); const app = express(); const server = http.createServer(app); const io = new Server(server); const db = new sqlite3.Database("./chat.db"); app.use(express.json()); /* ========================= SERVE FRONTEND ========================= */ app.get("/", (req, res) => { res.sendFile(path.join(__dirname, "index.html")); }); /* ========================= DB INIT ========================= */ db.serialize(() => { db.run(` CREATE TABLE IF NOT EXISTS messages ( id INTEGER PRIMARY KEY AUTOINCREMENT, room TEXT, sender TEXT, message TEXT, createdAt DATETIME DEFAULT CURRENT_TIMESTAMP ) `); }); /* ========================= STATE MANAGEMENT ========================= */ const onlineUsers = {}; // username -> socket.id const userSockets = {}; // username -> socket /* ========================= SOCKET.IO ========================= */ io.on("connection", (socket) => { console.log("User connected:", socket.id); /* ------------------------- LOGIN USER ------------------------- */ socket.on("login", (username) => { socket.username = username; onlineUsers[username] = true; userSockets[username] = socket; socket.join(username); // private room per user io.emit("onlineUsers", Object.keys(onlineUsers)); }); /* ------------------------- PRIVATE + GROUP MESSAGE ------------------------- */ socket.on("sendMessage", ({ roomId, sender, message }) => { // simpan chat db.run( `INSERT INTO messages (room, sender, message) VALUES (?, ?, ?)`, [roomId, sender, message] ); // kirim ke room io.to(roomId).emit("receiveMessage", { roomId, sender, message }); }); /* ------------------------- CREATE GROUP (SIMPLE UI) ------------------------- */ socket.on("createGroup", ({ groupName, members }) => { const roomId = "group_" + Date.now(); // join semua member ke room members.forEach(user => { const targetSocket = userSockets[user]; if (targetSocket) { targetSocket.join(roomId); } }); // broadcast group ke semua user io.emit("newGroup", { roomId, groupName, members }); }); /* ------------------------- DISCONNECT ------------------------- */ socket.on("disconnect", () => { const username = socket.username; if (username) { delete onlineUsers[username]; delete userSockets[username]; } io.emit("onlineUsers", Object.keys(onlineUsers)); }); }); /* ========================= CHAT HISTORY ========================= */ app.get("/messages/:roomId", (req, res) => { db.all( "SELECT * FROM messages WHERE room = ? ORDER BY createdAt ASC", [req.params.roomId], (err, rows) => { if (err) return res.status(500).send(err); res.json(rows); } ); }); /* ========================= START SERVER ========================= */ server.listen(3000, () => { console.log("Server running at http://localhost:3000"); });
